create or replace function public.leaderboard_competities( p_periode text default 'maand' ) returns table ( gebruikersnaam text, competities_gespeeld bigint, gewonnen bigint, verloren bigint, gelijkspel bigint, winpercentage numeric ) language plpgsql stable security definer set search_path = '' as $$ declare v_start timestamptz; v_einde timestamptz; begin if p_periode is null or p_periode not in ('maand', 'overall') then raise exception 'Ongeldige periode'; end if; if p_periode = 'maand' then v_start := ( date_trunc( 'month', now() at time zone 'Europe/Amsterdam' ) at time zone 'Europe/Amsterdam' ); v_einde := ( ( date_trunc( 'month', now() at time zone 'Europe/Amsterdam' ) + interval '1 month' ) at time zone 'Europe/Amsterdam' ); end if; return query with afgeronde_competities as ( select c.id from public.competities c join public.competitie_deelnemers d on d.competitie_id = c.id where c.start_op is not null and c.eindigt_op is not null and ( c.status = 'afgelopen' or c.eindigt_op <= now() ) and ( p_periode = 'overall' or ( c.eindigt_op >= v_start and c.eindigt_op < v_einde ) ) group by c.id having count(*) = 2 and count(*) filter ( where d.status = 'geaccepteerd' ) = 2 ), scores as ( select a.id as competitie_id, d.gebruiker_id, coalesce( sum(v.punten_verdiend) filter ( where v.geannuleerd_op is null ), 0 ) as punten from afgeronde_competities a join public.competitie_deelnemers d on d.competitie_id = a.id left join public.voltooide_klusjes v on v.competitie_id = a.id and v.gebruiker_id = d.gebruiker_id group by a.id, d.gebruiker_id ), uitslagen as ( select competitie_id, gebruiker_id, punten, max(punten) over ( partition by competitie_id ) as hoogste_score, min(punten) over ( partition by competitie_id ) as laagste_score from scores ), statistieken as ( select gebruiker_id, count(*) as gespeeld, count(*) filter ( where punten = hoogste_score and hoogste_score > laagste_score ) as winst, count(*) filter ( where punten = laagste_score and hoogste_score > laagste_score ) as verlies, count(*) filter ( where hoogste_score = laagste_score ) as gelijk from uitslagen group by gebruiker_id ) select p.gebruikersnaam, coalesce(s.gespeeld, 0)::bigint, coalesce(s.winst, 0)::bigint, coalesce(s.verlies, 0)::bigint, coalesce(s.gelijk, 0)::bigint, case when coalesce(s.winst, 0) + coalesce(s.verlies, 0) = 0 then null else round( 100.0 * s.winst / (s.winst + s.verlies), 1 ) end from public.profiles p left join statistieken s on s.gebruiker_id = p.id where p.gebruikersnaam is not null order by coalesce(s.winst, 0) desc, p.gebruikersnaam asc; end; $$; -- De functie publiek beschikbaar maken revoke all on function public.leaderboard_competities(text) from public; grant execute on function public.leaderboard_competities(text) to anon, authenticated;
